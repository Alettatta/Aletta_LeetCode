# Problem Description
给你二叉树的根节点 root，返回其节点值的层序遍历结果。即逐层地、从左到右访问所有节点。


# Code

## ACM version

```python
import sys 
from collections import deque

class TreeNode:
    def __init__(self, val = 0, left = None, right = None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def levelOrder(self, root):
        if root is None:
            return []
            
        ans = []
        q = deque([root])

        while q:
            size = len(q)
            level = []
            for _ in range(size):
                cur = q.popleft()
                level.append(cur.val)
                if cur.left is not None:
                    q.append(cur.left)
                if cur.right is not None:
                    q.append(cur.right)
            ans.append(level)
            
        return ans


def build_tree(s):
    data = s.strip().split() # ['3','9'……]
    if not data:                    # ✅ 空序列
        return None
    if data[0] == '#':              # ✅ 根为空
        return None
        
    root = TreeNode(int(data[0]))
    i = 1
    q = deque([root])
    
    while q and i < len(data):
        node = q.popleft()
        if i < len(data):
            t = data[i]; i += 1
            if t != '#':
                node.left = TreeNode(int(t))
                q.append(node.left)
        if i < len(data):
            t = data[i]; i += 1
            if t != '#':
                node.right = TreeNode(int(t))
                q.append(node.right)
                
    return root


data = sys.stdin.read().strip().split('\n')
idx = 0
while idx < len(data):
    length = int(data[idx]); idx += 1
    
    sequence = data[idx] if idx < len(data) else ''; idx += 1 # 读下一行作为层序序列（可能是空行）
    root = build_tree(sequence)
    
    result = Solution().levelOrder(root) # [[3],[9,20]]
    for level in result:
         print(' '.join(str(num) for num in level))
```

# Complexity Analysis
- 时间复杂度：O(n)，n 表示二叉树的节点总数，每个节点恰好被访问一次，每个节点出队时执行的操作都是 O(1) 的常数时间
- 空间复杂度：O(n)
