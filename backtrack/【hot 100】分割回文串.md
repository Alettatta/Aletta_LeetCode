# Problem Description
给你一个字符串 s，请你将 s 分割成一些子串，使每个子串都是 回文串。返回 s 所有可能的分割方案。


# Code

## ACM version

```python
import sys


# 有重复元素不可以复用，子集问题
class Solution:
    def partition(self, s):
        if not s:
            return []

        ans = []
        path = []

        def is_palindrome(sub):
            return sub == sub[::-1]

        def backtrack(start): # start：上一刀切完后的位置（包含）；i：当前这一刀要切到哪（包含）
            # base case：因为要把s切割；切到字符串末尾，说明完成一种分割方案
            if start == len(s):
                ans.append(path[:])
                return 
                
            for i in range(start, len(s)):
                sub = s[start:i + 1] # 切一刀使得子串为下标 start 到 i（含 i）的字符
                if not is_palindrome(sub):
                    continue
                path.append(sub)
                backtrack(i + 1)
                path.pop()

        backtrack(0)
        return ans


data = sys.stdin.read().strip().split('\n')
idx = 0
while idx < len(data):
    s = data[idx]; idx += 1
    ans = Solution().partition(s)
    print(len(ans))
    for ls in ans:
        print(' '.join(ls))
```

- 时间复杂度：O(n)
- 空间复杂度：O(n)
