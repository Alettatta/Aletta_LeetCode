# Problem Description
将两个升序链表合并为一个新的升序链表并返回。新链表是通过拼接给定的两个链表的所有节点组成的。

数据范围：

两个链表的节点数目范围都是 [0, 50]

-100 <= Node.val <= 100

list1 和 list2 均按 非递减顺序 排列

将两个升序链表 list1 和 list2 合并为一个新的升序链表并返回。


# Key Points
如果需要新建新链表就建立虚拟头结点；重点在于while循环的条件: 是p1不是p1.next


# Code

## ACM Version

```python
import sys

# ===== 1. 手写链表节点类 =====
class ListNode:
    def __init__(self, val: int = 0, next = None):
        self.val = val
        self.next = next

# ===== 2. 题解核心函数 =====
class Solution:
    def mergeTwoLists(self, list1: Optional[ListNode], list2: Optional[ListNode]) -> Optional[ListNode]:
        # 两个链表已经升序，因为要创建新的链表（新链表头结点未知）、所以使用dummy
        dummy = ListNode(-1)
        p = dummy # 新链表的指针
        p1 = list1 # list1和list2本身就是节点类型
        p2 = list2

        while p1 is not None and p2 is not None: # 注意不从p1.next开始因为list1和2的头结点还没遍历到
            if p1.val > p2.val:
                p.next = p2
                p2 = p2.next
            else:
                p.next = p1
                p1 = p1.next
            p = p.next

        # 如果两条链不一样长，会有剩余
        if p1 is not None:
            p.next = p1
        if p2 is not None:
            p.next = p2

        return dummy.next    

# ===== 3. 由数组构造链表 =====
def build_list(arr):
    dummy = ListNode(-1)
    p = dummy
    for x in arr:
        p.next = ListNode(x)
        p = p.next
    return dummy.next

# ===== 4. 打印链表 =====
def print_list(head):
    vals = []
    p = head
    while p:
        vals.append(str(p.val))
        p = p.next
    print(len(vals), " ".join(vals))

# ===== 5. ACM 输入输出 =====
def main():
    data = sys.stdin.read().strip().split('\n')
    idx = 0
    while idx + 1 < len(data): # 输入一次必须2行，保证2行都存在
        parts = data[idx].split(); idx += 1
        len1 = int(parts[0])
        array1 = list(map(int, parts[1: 1 + len1]))

        parts = data[idx].split(); idx += 1
        len2 = int(parts[0])
        array2 = list(map(int, parts[1:1 + len2]))

        list1 = build_list(array1)
        list2 = build_list(array2)
        ans = Solution().mergeTwoLists(list1, list2)
        print_list(ans)

if __name__ == "__main__":
    main()
```
