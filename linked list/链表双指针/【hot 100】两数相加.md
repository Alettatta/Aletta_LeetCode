# Problem Description
给你两个非空的链表，表示两个非负的整数。它们每位数字都是按照逆序的方式存储的，并且每个节点只能存储一位数字。

请你将两个数相加，并以相同形式返回一个表示和的链表。

你可以假设除了数字 0 之外，这两个数都不会以 0 开头。

给你两个非空链表 l1 和 l2，每个节点存储一位数字，整数按逆序存储（即个位在链表头部）。请你计算这两个整数的和，并以相同的形式（逆序、每个节点一位数字）返回一个新链表。


# Key Points
处理进位：每一位是 p1.val + p2.val + carry（进位）；当前位结果 = total % 10；进位 = total // 10（取整）

# Code
## ACM version

```python
# 处理进位：每一位是 p1.val + p2.val + carry（进位）
# 当前位结果 = total % 10；进位 = total // 10（取整）

import sys

# 链表类
class ListNode:
    def __init__(self, val = 0, next = None):
        self.val = val
        self.next = next

# 核心函数
class Solution:
    def addTwoNumbers(self, l1, l2):
        dummy = ListNode(-1)
        p1, p2 = l1, l2
        p = dummy
        carry = 0

        while p1 is not None or p2 is not None or carry > 0:
            v1 = p1.val if p1 is not None else 0
            v2 = p2.val if p2 is not None else 0

            total = v1 + v2 + carry
            carry = total // 10 # 进位
            p.next = ListNode(total % 10) # 当前位
            p = p.next

            if p1 is not None:
                p1 = p1.next
            if p2 is not None:
                p2 = p2.next

        return dummy.next

# 由数组构造链表
def build_list(array):
    dummy = ListNode(-1)
    p = dummy
    for x in array:
        p.next = ListNode(x)
        p = p.next
    return dummy.next
        
# 打印链表
def print_list(head):
    vals = []
    p = head
    while p is not None:
        vals.append(str(p.val))
        p = p.next
    print(len(vals), ' '.join(vals))

# 主函数
def main():
    data = sys.stdin.read().strip().split('\n')
    idx = 0
    while idx + 1 < len(data): # 输入一次必须2行，保证2行都存在
        parts = data[idx].split(); idx += 1
        len1 = int(parts[0])
        array1 = list(map(int, parts[1: 1 + len1]))

        parts = data[idx].split(); idx += 1
        len2 = int(parts[0])
        array2 = list(map(int, parts[1:1 + len2]))

        list1 = build_list(array1)
        list2 = build_list(array2)
        ans = Solution().addTwoNumbers(list1, list2)
        print_list(ans)
        
if __name__ == "__main__":
    main()
```
